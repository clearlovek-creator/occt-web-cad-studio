cmake_minimum_required(VERSION 3.20)
project(cad_patch_volume_mesh_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GMSH_INCLUDE_DIR "" CACHE PATH "Verified Gmsh API header directory")
set(GMSH_LIBRARY "" CACHE FILEPATH "Verified Gmsh import library")
if(NOT GMSH_INCLUDE_DIR OR NOT GMSH_LIBRARY)
  message(FATAL_ERROR "Explicit Gmsh include and library are required")
endif()
add_subdirectory(../../mesh-worker/src/discrete "${CMAKE_CURRENT_BINARY_DIR}/discrete-core")
add_executable(cad_patch_volume_mesh_research main.cxx ../../geometry-worker/src/protocol/Json.cxx ../../geometry-worker/src/runtime/NativeResourceGuard.cxx)
target_link_libraries(cad_patch_volume_mesh_research PRIVATE cad_discrete_volume_mesh_core)
if(MINGW)
  target_link_options(cad_patch_volume_mesh_research PRIVATE -static-libgcc -static-libstdc++)
endif()
